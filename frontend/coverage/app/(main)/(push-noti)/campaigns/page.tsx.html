                ))
              )}
            </tbody>
          </table>
        </div>
      </div>
 
      {/* Modal para crear campaña */}
      <CreateCampaignModal
        isOpen={isCreateModalOpen}
        onClose={() => setIsCreateModalOpen(false)}
        onSuccess={handleCampaignCreated}
      />
    </div>
  );
}
